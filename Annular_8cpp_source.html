122template <class Real>
123void Annular<Real>::InterpR(sctl::Vector<Real>& trg_r, const sctl::Vector<Real> src_r, const sctl::Vector<Real> src_x, const sctl::Vector<Real> trg_x) {
124 sctl::Matrix<Real> Minterp(src_x.Dim(), trg_x.Dim());
125 sctl::Vector<Real> wts(src_x.Dim()*trg_x.Dim(), (sctl::Iterator<Real>)Minterp.begin(), false);
126 sctl::LagrangeInterp<Real>::Interpolate(wts, src_x, trg_x); // row-major order, Ns x Nt (stacked) so wts for all targets from first source first, from second source, etc.
127 // left multiply by source value of r to get row of target r values.
128 SCTL_ASSERT(trg_r.Dim() == trg_x.Dim());
129 sctl::Matrix<Real> trg_r_mat(1, trg_r.Dim(), (sctl::Iterator<Real>)trg_r.begin(),false);
130 sctl::Matrix<Real>::GEMM(trg_r_mat, sctl::Matrix<Real>(1,src_r.Dim(),(sctl::Iterator<Real>)src_r.begin(),false), Minterp);
131}

479template <class Real>
480sctl::Vector<Real> Annular<Real>::GetCenterLine(sctl::Long Nelem_, sctl::Long ElemOrder_) {
481 // Returns centerline nodes when x=[0,1] divided into <Nelem> panels of <ElemOrder> number of Cheb nodes each
482 const sctl::Vector<Real> nodes_01 = sctl::SlenderElemList<Real>::CenterlineNodes(ElemOrder_);
483 SCTL_ASSERT(ElemOrder_ == nodes_01.Dim());
484 sctl::Vector<Real> Xc(ElemOrder_ * Nelem_ * 3);
485 for (sctl::Long elem_ind = 0; elem_ind < Nelem_; elem_ind++) {
486 for (sctl::Long node_ind = 0; node_ind < ElemOrder_; node_ind++) {
487 const Real x = (elem_ind + nodes_01[node_ind]) / Nelem_;
488 Xc[(elem_ind*ElemOrder_+node_ind)*3 + 0] = x;
489 Xc[(elem_ind*ElemOrder_+node_ind)*3 + 1] = 0.5;
490 Xc[(elem_ind*ElemOrder_+node_ind)*3 + 2] = 0.5;
491 }
492 }
493 return Xc;

496template <class Real>
497std::tuple<sctl::Long, sctl::Long, sctl::Vector<Real>> Annular<Real>::GetCenterLine_mpi(sctl::Long Nelem_, sctl::Long ElemOrder_, sctl::Comm comm) {
498 sctl::Long Nmpi = comm.Size();
499 sctl::Long loc_elem_cnt = Nelem_ / Nmpi;
500 sctl::Long remainder = Nelem_ - loc_elem_cnt * Nmpi;
501 sctl::Long loc_elem_dsp;
502 if (comm.Rank() < remainder) {
503 // if rank smaller than remainder from division, add a panel
504 loc_elem_cnt += 1;
505 loc_elem_dsp = comm.Rank() * loc_elem_cnt;
506 } else {
507 loc_elem_dsp = comm.Rank() * loc_elem_cnt + remainder;
508 }
509
510 const sctl::Vector<Real> nodes_01 = sctl::SlenderElemList<Real>::CenterlineNodes(ElemOrder_);
511 sctl::Vector<Real> Xc(ElemOrder_ * loc_elem_cnt * 3);
512 Xc = 0.5;
513 for (sctl::Long elem_ind = loc_elem_dsp; elem_ind < loc_elem_dsp + loc_elem_cnt; elem_ind++) {
514 for (sctl::Long node_ind = 0; node_ind < ElemOrder_; node_ind++) {
515 const Real x = (elem_ind + nodes_01[node_ind]) / Nelem_;
516 Xc[((elem_ind - loc_elem_dsp)*ElemOrder_ + node_ind)*3 + 0] = x;
517 }
518 }
519
520 return std::make_tuple(loc_elem_cnt, loc_elem_dsp, Xc);
521}

529template <class Real>
530sctl::Vector<sctl::Long> Annular<Real>::InDomain(sctl::Vector<Real> X_) {
531 sctl::Long Nnodes = X_.Dim()/3;
532 sctl::Vector<sctl::Long> in_domain(Nnodes);
533 for (sctl::Long nd = 0; nd < Nnodes; nd++) {
534 const Real x = X_[nd*3+0];
535 const Real y = X_[nd*3+1];
536 const Real z = X_[nd*3+2];
537 if (x < Xc_inner[0] || x < Xc_outer[0] || x > Xc_inner[Xc_inner.Dim()-3] || x > Xc_outer[Xc_outer.Dim()-3]) {
538 std::cout << "x value out of bounds , don't consider." << std::endl;
539 in_domain[nd] = 0;
540 } else {
541 const auto search_panel = [](Real x, sctl::Long Nelem_, sctl::Long ElemOrder_, sctl::Vector<Real> Xc_) {
542 for (sctl::Long ind=0; ind<Nelem_; ind++) {
543 if (x < Xc_[(ind+1)*ElemOrder_*3 - 3]) {
544 // x smaller than right end of this interval, return element index.
545 return ind;
546 }
547 }
548 std::cout << "Did not find panel, something went wrong." << std::endl;
549 return static_cast<sctl::Long>(-1);
550 };
551 // find panel x belongs to
552 sctl::Long x_inner_ind = search_panel(x, Nelem_inner, ElemOrder_inner, Xc_inner);
553 sctl::Long x_outer_ind = search_panel(x, Nelem_outer, ElemOrder_outer, Xc_outer);
554 // interpolate r to this x value
555 sctl::Vector<Real> r_inner_interp(1);
556 sctl::Vector<Real> src_r(ElemOrder_inner, (sctl::Iterator<Real>)r_inner.begin()+x_inner_ind*ElemOrder_inner, false);
557 sctl::Vector<Real> src_x(ElemOrder_inner);
558 for (sctl::Long node_ind = 0; node_ind < ElemOrder_inner; node_ind++) {
559 src_x[node_ind] = Xc_inner[x_inner_ind*ElemOrder_inner*3 + node_ind*3];
560 }
561 sctl::Vector<Real> trg_x(1, (sctl::Iterator<Real>)X_.begin() + 3*nd, false);
562 InterpR(r_inner_interp, src_r, src_x, trg_x);
563
564 sctl::Vector<Real> r_outer_interp(1);
565 src_x.SetZero();
566 for (sctl::Long node_ind = 0; node_ind < ElemOrder_outer; node_ind++) {
567 src_x[node_ind] = Xc_outer[x_outer_ind*ElemOrder_outer*3 + node_ind*3];
568 }
569 sctl::Vector<Real> src_r2(ElemOrder_outer, (sctl::Iterator<Real>)r_outer.begin()+x_outer_ind*ElemOrder_outer, false);
570 InterpR(r_outer_interp, src_r2, src_x, trg_x);
571 // compare r^2 with (y^2+z^2)
572 in_domain[nd] = ((r_outer_interp[0]*r_outer_interp[0] - (y-0.5)*(y-0.5) - (z-0.5)*(z-0.5)) > 1e-5) && ((y-0.5)*(y-0.5) + (z-0.5)*(z-0.5) - (r_inner_interp[0]*r_inner_interp[0]) > 1e-5);
573 }
574 }
575 return in_domain;
576}

673template <class Real>
674sctl::Vector<Real> Annular<Real>::GetVslipInner() {
675 SCTL_ASSERT(SetupInner_bool); // check that inner channel is set up.
676 // r vector for each angular node, per center node.
677 sctl::Vector<Real> vslip(X_inner.Dim());
678 SCTL_ASSERT(vslip.Dim() == Nelem_inner*ElemOrder_inner*FourierOrder_inner * 3);
679
680 for (sctl::Long cn=0; cn < Nelem_inner * ElemOrder_inner; cn++) {
681 sctl::Vector<Real> Xc_here(3, (sctl::Iterator<Real>) Xc_inner.begin() + cn*3, false); // iterator starting at the x coord of current center node.
682 for (sctl::Long fn = 0; fn < FourierOrder_inner; fn++) { // check that this is the number of F nodes
683 sctl::Long fncn = 3*(cn * FourierOrder_inner + fn);
684 sctl::Vector<Real> X_here(3, (sctl::Iterator<Real>) X_inner.begin() + fncn, false);
685 sctl::Vector<Real> rvec_here = X_here - Xc_here;
686 Real rnorm = rvec_here[0]*rvec_here[0] + rvec_here[1]*rvec_here[1] + rvec_here[2]*rvec_here[2];
687 rnorm = sctl::sqrt<Real>(rnorm);
688 sctl::Vector<Real> vslip_here(3, (sctl::Iterator<Real>) vslip.begin() + fncn, false);
689 vslip_here = drdt_inner[cn] / rnorm * rvec_here; // vslip points out in r direction with magnitude drdt.
690 }
691 }
692 return vslip;
693}
