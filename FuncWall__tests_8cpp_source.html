7template <class Real>
8void set_centerline_x_linspace(sctl::Vector<Real>& coords,
9 Real x_start,
10 Real x_end,
11 Real y_fixed,
12 Real z_fixed) {
13 const sctl::Long dim = coords.Dim();
14 SCTL_ASSERT(dim % 3 == 0);
15 const sctl::Long N = dim / 3;
16
17 if (N <= 0) return;
18 if (N == 1) {
19 coords[0] = x_start;
20 coords[1] = y_fixed;
21 coords[2] = z_fixed;
22 return;
23 }
24
25 const Real dx = (x_end - x_start) / (Real)(N - 1);
26
27 for (sctl::Long i = 0; i < N; ++i) {
28 Real x = x_start + dx * (Real)i;
29 coords[3*i ] = x;
30 coords[3*i + 1] = y_fixed;
31 coords[3*i + 2] = z_fixed;
32 }
33}

74// 3. Spatiotemporal Functor (4 Arguments: Radius, Rdot, Time, Coords)
75// Radius varies based on the x-coordinate: R(x) = Base + Slope * x + Amplitude * sin(omega * t)
76template <class Real>
77struct SpatiotemporalFunctor {
78 Real _base;
79 Real _slope;
80 Real _amplitude;
81 Real _omega;
82 SpatiotemporalFunctor(Real base, Real slope, Real amplitude, Real omega)
83 : _base(base), _slope(slope), _amplitude(amplitude), _omega(omega) {}
84 // This signature triggers the 'if constexpr' true branch in FuncWall
85 void operator()(sctl::Vector<Real>& radius, sctl::Vector<Real>& rdot, Real time, const sctl::Vector<Real>& coords) const {
86 Real sin_term = std::sin(_omega * time);
87 Real cos_term = std::cos(_omega * time);
88 for (sctl::Long i = 0; i < radius.Dim(); ++i) {
89 // Coords are packed: [x0, y0, z0, x1, y1, z1, ...]
90 Real x_coord = coords[i * 3];
91 radius[i] = _base + _slope * x_coord + _amplitude * sin_term;
92 rdot[i] = _amplitude * _omega * cos_term;
93 }
94 }
95};

97// 4. r^2 spatiotemporal Functor (4 Arguments: Radius, Rdot, Time, Coords)
98// Radius varies based on the x-coordinate: r(x,t) = R0*sqrt(1 + Amplitude * cos( 2*pi * n* x/L) * cos(omega * t))
99// Rdot is computed accordingly dr/dt = - (R0 * Amplitude * omega * cos(2*pi*n*x/L) * sin(omega*t)) / (2*sqrt(1 + Amplitude * cos(2*pi*n*x/L) * cos(omega*t)))
100template <class Real>
101struct R2SpatiotemporalFunctor {
102 Real _R0;
103 Real _amplitude;
104 Real _L;
105 Real _omega;
106 sctl::Long _n;
107 R2SpatiotemporalFunctor(Real R0, Real amplitude, Real L, sctl::Long n, Real omega)
108 : _R0(R0), _amplitude(amplitude), _L(L), _n(n), _omega(omega) {}
109 // This signature triggers the 'if constexpr' true branch in FuncWall
110 void operator()(sctl::Vector<Real>& radius, sctl::Vector<Real>& rdot, Real time, const sctl::Vector<Real>& coords) const {
111 Real cos_omega_t = std::cos(_omega * time);
112 Real sin_omega_t = std::sin(_omega * time);
113 for (sctl::Long i = 0; i < radius.Dim(); ++i) {
114 // Coords are packed: [x0, y0, z0, x1, y1, z1, ...]
115 Real x_coord = coords[i * 3];
116 Real cos_term = std::cos(2.0 * sctl::const_pi<Real>() * _n * x_coord / _L);
117 Real radicand = 1.0 + _amplitude * cos_term * cos_omega_t;
118 radius[i] = _R0 * std::sqrt(radicand);
119 // Derivative calculation
120 if (radicand > 0.0) {
121 rdot[i] = - (_R0 * _amplitude * _omega * cos_term * sin_omega_t) / (2.0 * std::sqrt(radicand));
122 } else {
123 rdot[i] = 0.0; // Avoid NaN, though this indicates an issue
124 }
125 }
126 }
127};
