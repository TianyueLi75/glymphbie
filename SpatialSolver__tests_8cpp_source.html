9// Solve bg poiseuille flow -- dpdx = p(x=1)-p(x=0) = negative for flow right
10template <class Real>
11sctl::Vector<Real> bg_poiseuille(const Real dpdx, const Real mu, const sctl::Vector<Real> X)
12{
13 const sctl::Long N = X.Dim()/3;
14 sctl::Vector<Real> U(N*3);
15 U.SetZero();
16 for (sctl::Long i = 0; i < N; i++) {
17 sctl::Vector<Real> x(3, (sctl::Iterator<Real>) X.begin() + i*3, false);
18 U[i*3+0] = dpdx * ((x[1]-0.5)*(x[1]-0.5) + (x[2]-0.5)*(x[2]-0.5))/4./mu;
19 }
20 return U;
21}

37template <class Real>
38void concentric_poiseuille(const Real dpdx, const Real mu, const sctl::Long Nelem, const sctl::Long ElemOrder, const sctl::Long FourierOrder, sctl::Comm comm)
39{
40 const Real R_in = 0.3;
41 const Real R_out = 0.48;
42
43 // For Fourier order that is too small, GMRES will plateau early, so a larger tolerance will suffice.
44 Real gmres_tol = 1e-11;
45 Real tol = 1e-14;
46 if (FourierOrder < 16) {
47 gmres_tol = 1e-9;
48 }
49
50 // ==========================
51 // Make Annular channel
52 // ==========================
53 // Set centerline x values that are <ElemOrder>-order Chebyshev nodes
54 sctl::Vector<Real> Xc = Annular<Real>::GetCenterLine(Nelem, ElemOrder);
55 // Set radius and wall velocity parameters
56 sctl::Vector<Real> r1(ElemOrder*Nelem);
57 r1 = R_in;
58 sctl::Vector<Real> r2(ElemOrder*Nelem);
59 r2 = R_out;
60 sctl::Vector<Real> drdt(ElemOrder*Nelem); // no outward velocity on wall.
61 drdt = 0.;
62 // Create annular object, concentric with different radii
63 Annular<Real> straight(Xc,Xc,r1,r2);
64 // Set up annular object using <FourierOrder> nodes in azimuthal direction.
65 straight.Setup(Nelem, ElemOrder, FourierOrder,drdt,drdt);
66 // Get surface nodes on both channels.
67 sctl::Vector<Real> X_annular;
68 straight.GetNodeCoord(&X_annular, nullptr);
69
70 // ==========================
71 // Exact solution
72 // ==========================
73 // NOTE: This assumes Xvec taken as an intermediate channel with the same parameters for simplicity.
74 // Otherwise an interpolation is needed.
75 const auto uexact = [R_in, R_out, dpdx, mu, &Xc, Nelem, ElemOrder, FourierOrder](const sctl::Vector<Real> Xvec) {
76 sctl::Long N = Xvec.Dim()/3;
77 SCTL_ASSERT(Xvec.Dim() == Nelem * ElemOrder * FourierOrder * 3);
78 sctl::Vector<Real> Uvec(Xvec.Dim());
79 Uvec.SetZero();
80 // compute constants first
81 Real coeff = 1./4./mu;
82 // Get rmax
83 Real rmax_num = R_out*R_out - R_in*R_in;
84 Real Rratio = R_out / R_in;
85 Real rmax_denom = 2*sctl::log<Real>(Rratio);
86 Real rmax2 = rmax_num / rmax_denom;
87 // Compute u = 1/4/mu*dpdx*(R_out^2-r^2-2rmax^2ln*(R_out/r))
88 for (sctl::Long i=0; i<Nelem; i++) {
89 for (sctl::Long j=0; j<ElemOrder; j++) {
90 // get Xc at this index
91 sctl::Vector<Real> Xc_here(3, (sctl::Iterator<Real>)Xc.begin() + (i*ElemOrder+j)*3, false);
92 for (sctl::Long k=0; k<FourierOrder; k++) {
93 sctl::Long starting_ind = (i*ElemOrder*FourierOrder + j*FourierOrder + k)*3;
94 // Get node coordinates at this index
95 sctl::Vector<Real> Xhere(3, (sctl::Iterator<Real>) Xvec.begin() + starting_ind, false);
96 sctl::Vector<Real> Rhere = Xhere - Xc_here;
97 Real r2 = Rhere[1]*Rhere[1] + Rhere[2]*Rhere[2];
98 Real rratio = R_out / sctl::sqrt<Real>(r2);
99 Real u = -1.*coeff * dpdx * (R_out*R_out - r2 - 2. * rmax2 * sctl::log<Real>(rratio));
100 Uvec[starting_ind + 0] = u; // flow always just in the x direction.
101 }
102 }
103 }
104 return Uvec;
105 };
106
107 // ==========================
108 // StokesBIO
109 // ==========================
110 // Real min_rad = straight.GetMinRadius();
111 // Real S_scal = 1./(2.*min_rad*sctl::log<Real>(1./min_rad)); // TODO: stable computation? check csbq code.
112
113 // Use a simple D+S operator, which is full-rank.
114 Real S_scal = 1.0;
115 Real D_scal = 1.0;
116 StokesBIO<Real> LPO(S_scal, D_scal, comm);
117 // Add inner and outer channels to LPO
118 LPO.AddElemList(straight.GetInnerElemList(), "inner");
119 LPO.AddElemList(straight.GetOuterElemList(), "outer");
120 // Set tolerance and periodicity for PVFMM
121 LPO.SetAccuracy(tol);
122 LPO.SetPeriodicity(sctl::Periodicity::X, 1.0);
123 // Set target coordinates and source normal
124 LPO.SetTargetCoord(X_annular);
125 sctl::Vector<Real> NormalOrient(X_annular.Dim());
126 NormalOrient = 1.;
127 sctl::Long size_inner = Nelem*ElemOrder*FourierOrder*3;
128 sctl::Long size_outer = X_annular.Dim() - size_inner;
129 for (sctl::Long ind=0; ind<size_outer; ind++) {
130 // Normal of outer channel element list points outward by default.
131 NormalOrient[size_inner + ind] = -1.;
132 }
133
134 const auto BIO = [&LPO, D_scal, &NormalOrient](sctl::Vector<Real>* U, const sctl::Vector<Real> sigma) {
135 U->SetZero();
136 LPO.ComputePotential(*U, sigma);
137 if (D_scal && U->Dim() == sigma.Dim()) (*U) += 0.5*sigma*D_scal*NormalOrient; // double layer jump condition on surface.
138 };
139
140 // ==========================
141 // GMRES solve
142 // ==========================
143 sctl::GMRES<Real> solver(comm);
144 sctl::KrylovPrecond<Real> ksp;
145 sctl::Vector<Real> sigma;
146 sctl::Vector<Real> vslip = straight.GetVslip(); // wall velocity from dr/dt
147 sctl::Vector<Real> vbg = bg_poiseuille(dpdx,mu,X_annular); // background flow
148 solver(&sigma, BIO, vslip-vbg, gmres_tol, -1, false, nullptr, &ksp);
149
150 // ==========================
151 // Error in flow U
152 // ==========================
153 // For a easy set of target points, use surface nodes on two cylinders in between.
154 sctl::Vector<Real> r3(ElemOrder*Nelem);
155 r3 = R_in + 1./3. * (R_out - R_in);
156 sctl::Vector<Real> r4(ElemOrder*Nelem);
157 r4 = R_in + 2./3. * (R_out - R_in);
158 // Set up Annular object to grab surface nodes
159 Annular<Real> channel_trg(Xc,Xc,r3,r4);
160 channel_trg.Setup(Nelem, ElemOrder, FourierOrder,drdt,drdt);
161 sctl::Vector<Real> X_inner, X_outer;
162 channel_trg.GetInnerCoord(&X_inner, nullptr);
163 channel_trg.GetOuterCoord(&X_outer, nullptr);
164 // Evaluate at target
165 sctl::Vector<Real> U_inner, U_outer;
166 LPO.SetTargetCoord(X_inner);
167 BIO(&U_inner, sigma);
168 LPO.SetTargetCoord(X_outer);
169 BIO(&U_outer, sigma);
170 // Add background flow
171 U_inner += bg_poiseuille(dpdx,mu,X_inner);
172 U_outer += bg_poiseuille(dpdx,mu,X_outer);
173 // Get exact flow
174 sctl::Vector<Real> Uexact_inner = uexact(X_inner);
175 sctl::Vector<Real> Uexact_outer = uexact(X_outer);
176 // Gather max error at either radii, average to one relative error.
177 Real max_err_inner = 0.;
178 Real max_err_outer = 0.;
179 sctl::Vector<Real> Diff_inner = U_inner - Uexact_inner;
180 sctl::Vector<Real> Diff_outer = U_outer - Uexact_outer;
181 Real max_u_inner = 0.;
182 Real max_u_outer = 0.;
183 for (auto e : U_inner) max_u_inner = std::max<Real>(max_u_inner, sctl::fabs(e));
184 for (auto e : U_outer) max_u_outer = std::max<Real>(max_u_outer, sctl::fabs(e));
185 for (auto e : Diff_inner) max_err_inner = std::max<Real>(max_err_inner, sctl::fabs(e));
186 for (auto e : Diff_outer) max_err_outer = std::max<Real>(max_err_outer, sctl::fabs(e));
187 Real avg_max_err = 0.5*(max_err_inner / max_u_inner + max_err_outer / max_u_outer);
188 std::cout << "Nelem = " << Nelem << ", Fourier order = " << FourierOrder << "; Averaged max relative error at two different r: "<< avg_max_err << std::endl;
189
190 // ==========================
191 // Error in flux Q
192 // ==========================
193 const Real alpha = R_out / R_in;
194 const Real beta = alpha;
195 const Real K = alpha*beta - 1.0;
196 const Real Q_ana = sctl::const_pi<Real>()/8. *((K+1.)*(K+1.) - 1.0 - 2.*K*K/sctl::log<Real>(K+1.));
197
198 // Compute Q by doing integral over outlet flow
199 const auto getQ = [&LPO, &R_in, &R_out, dpdx, mu, &sigma, &BIO](const sctl::Long r_ord, const sctl::Long theta_ord) {
200 SCTL_ASSERT(dpdx<0.);
201
202 // Get Chebyshev node and weights for radial direction
203 sctl::Vector<Real> nds, wts;
204 sctl::ChebQuadRule<Real>::ComputeNdsWts(&nds, &wts, r_ord); // nds and wts on [0,1]
205 sctl::Vector<Real> r_nodes = nds * (R_out - R_in) + R_in;
206 sctl::Vector<Real> r_wts = wts * (R_out - R_in);
207 // Get equidistant nodes for azimuthal direction
208 sctl::Vector<Real> theta_nodes(theta_ord);
209 const Real theta_wt = 2*sctl::const_pi<Real>()/theta_ord;
210 for (sctl::Long i=0; i<theta_ord; i++) {
211 theta_nodes[i] = theta_wt * i;
212 }
213 // form mesh grid of x-y points, same-r-all-thetas order.
214 sctl::Vector<Real> Xtrg(3 * r_ord * theta_ord);
215 for (sctl::Long i=0; i<r_ord; i++) {
216 for (sctl::Long j=0; j<theta_ord; j++) {
217 sctl::Long node_ind = i*theta_ord + j;
218 Xtrg[3*node_ind + 0] = 1.0; // evaluation at x=1.0 only
219 Xtrg[3*node_ind + 1] = 0.5 + r_nodes[i] * sctl::cos<Real>(theta_nodes[j]);
220 Xtrg[3*node_ind + 2] = 0.5 + r_nodes[i] * sctl::sin<Real>(theta_nodes[j]);
221 }
222 }
223
224 // Compute U at meshgrid:
225 sctl::Vector<Real> Utrg;
226 LPO.SetTargetCoord(Xtrg);
227 BIO(&Utrg, sigma);
228 Utrg += bg_poiseuille(dpdx,mu,Xtrg);
229
230 // Quadrature to integrate
231 Real Q = 0.;
232 for (sctl::Long i=0; i<r_ord; i++) {
233 for (sctl::Long j=0; j<theta_ord; j++) {
234 sctl::Long node_ind = i*theta_ord + j;
235 Real u = Utrg[3*node_ind + 0];
236 Q += u * r_nodes[i] * theta_wt * r_wts[i];
237 }
238 }
239 Real Q2 = Q * mu / (-dpdx) / R_in / R_in / R_in / R_in; // Q in paper = mu/-dp/R_in^4 * int u_code dA
240 return Q2;
241 };
242
243 Real Q = getQ(20,64);
244
245 std::cout << "Flux difference: Q analytical = " << std::setprecision(10) << Q_ana << ", Q computed = " << Q << "; rel error is " << fabs(Q-Q_ana) / fabs(Q_ana) << std::endl;
246
247}

459// For eccentric channels, use fixed centerlines and radii to prevent intersection.
460template <class Real>
461void eccentric_poiseuille(const Real dpdx, const Real mu, const sctl::Long Nelem, const sctl::Long ElemOrder, const sctl::Long FourierOrder, sctl::Comm comm)
462{
463
464 Real gmres_tol = 1e-11;
465 Real tol = 1e-14;
466 if (FourierOrder < 16) {
467 gmres_tol = 1e-9;
468 }
469
470 const Real R_in = 0.13;
471 const Real R_out = 0.48;
472 const Real Xc_y_out = 0.5;
473 const Real Xc_z_out = 0.5;
474 const Real Xc_y_in = 0.6;
475 const Real Xc_z_in = 0.4;
476
477 // Compute parameters for analytical solutions
478 const Real alpha = R_out / R_in;
479 const Real beta = alpha;
480 const Real K = alpha*beta - 1.0;
481 const Real c = sctl::sqrt<Real>((Xc_y_out-Xc_y_in)*(Xc_y_out-Xc_y_in) + (Xc_z_out-Xc_z_in)*(Xc_z_out-Xc_z_in));
482 // std::cout << "check c, should be 0.1414: " << c << std::endl;
483
484 const Real alpha2 = alpha * alpha;
485 const Real alpha4 = alpha2 * alpha2;
486 const Real eccen = c / R_in;
487 const Real eccen2 = eccen * eccen;
488 const Real F = (alpha2 - 1 + eccen2)/2./eccen;
489 const Real M = sctl::sqrt<Real>((F*F-alpha2));
490 const Real A = 0.5*sctl::log<Real>((F+M)/(F-M));
491 const Real B = 0.5*sctl::log<Real>((F-eccen+M)/(F-eccen-M));
