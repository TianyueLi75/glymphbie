74template <class Real>
75void StokesBIO<Real>::ComputePotential(sctl::Vector<Real>& U, const sctl::Vector<Real>& F) const {
76 sctl::Vector<Real> Us, Ud;
77 if (SL_scal_) LayerPotenSL.ComputePotential(Us, F);
78 if (DL_scal_) LayerPotenDL.ComputePotential(Ud, F);
79
80 if (SL_scal_ && DL_scal_) U = Us * SL_scal_ + Ud * DL_scal_;
81 else if (SL_scal_) U = Us * SL_scal_;
82 else if (DL_scal_) U = Ud * DL_scal_;
83 else U.SetZero();
84}
