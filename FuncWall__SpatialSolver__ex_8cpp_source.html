67 // NOTE: This assumes Xvec taken as an intermediate channel with the same parameters for simplicity. Otherwise an interpolation is needed.
68 const auto uexact = [&wall, dpdx, mu, &Xc, Nelem, ElemOrder, FourierOrder](const sctl::Vector<Real> Xvec) {
69 Real R_in = wall.radiusIn()[0];
70 Real R_out = wall.radiusOut()[0];
71 sctl::Long N = Xvec.Dim()/3;
72 SCTL_ASSERT(Xvec.Dim() == Nelem * ElemOrder * FourierOrder * 3);
73 sctl::Vector<Real> Uvec(Xvec.Dim());
74 Uvec.SetZero();
75 // compute constants first -- TODO: arithemetic order to reduce numerical error
76 Real coeff = 1./4./mu;
77 // Get rmax
78 Real rmax_num = R_out*R_out - R_in*R_in;
79 Real Rratio = R_out / R_in;
80 Real rmax_denom = 2*sctl::log<Real>(Rratio);
81 Real rmax2 = rmax_num / rmax_denom;
82 // Compute u = 1/4/mu*dpdx*(R_out^2-r^2-2rmax^2ln*(R_out/r))
83 for (sctl::Long i=0; i<Nelem; i++) {
84 for (sctl::Long j=0; j<ElemOrder; j++) {
85 // get Xc at this index
86 sctl::Vector<Real> Xc_here(3, (sctl::Iterator<Real>)Xc.begin() + (i*ElemOrder+j)*3, false);
87 for (sctl::Long k=0; k<FourierOrder; k++) {
88 sctl::Long starting_ind = (i*ElemOrder*FourierOrder + j*FourierOrder + k)*3;
89 sctl::Vector<Real> Xhere(3, (sctl::Iterator<Real>) Xvec.begin() + starting_ind, false);
90 sctl::Vector<Real> Rhere = Xhere - Xc_here;
91 Real r2 = Rhere[1]*Rhere[1] + Rhere[2]*Rhere[2];
92 Real rratio = R_out / sctl::sqrt<Real>(r2);
93 Real u = -1.*coeff * dpdx * (R_out*R_out - r2 - 2. * rmax2 * sctl::log<Real>(rratio));
94 Uvec[starting_ind + 0] = u; // flow always just in the x direction.
95 }
96 }
97 }
98 return Uvec;
99 };

101 // Solve bg poiseuille flow -- dpdx = p(x=1)-p(x=0) = negative for flow right
102 const auto bg_poiseuille = [dpdx, mu](const sctl::Vector<Real> X) {
103 const sctl::Long N = X.Dim()/3;
104 sctl::Vector<Real> U(N*3);
105 U.SetZero();
106 for (sctl::Long i = 0; i < N; i++) {
107 sctl::Vector<Real> x(3, (sctl::Iterator<Real>) X.begin() + i*3, false);
108 U[i*3+0] = dpdx * ((x[1]-0.5)*(x[1]-0.5) + (x[2]-0.5)*(x[2]-0.5))/4./mu;
109 }
110 return U;

218 // Compute Q by doing integral over outlet
219 Real Rin = wall.radiusIn()[0];
220 Real Rout = wall.radiusOut()[0];
221 const auto getQ = [&LPO, &Rin, &Rout, dpdx, mu, &sigma, &BIO, &bg_poiseuille](const sctl::Long r_ord, const sctl::Long theta_ord) {
222 SCTL_ASSERT(dpdx<0.);
223
224 sctl::Vector<Real> nds, wts;
225 sctl::ChebQuadRule<Real>::ComputeNdsWts(&nds, &wts, r_ord); // nds and wts on [0,1]
226 sctl::Vector<Real> r_nodes = nds * (Rout - Rin) + Rin;
227 sctl::Vector<Real> r_wts = wts * (Rout - Rin);
228
229 sctl::Vector<Real> theta_nodes(theta_ord);
230 const Real theta_wt = 2*sctl::const_pi<Real>()/theta_ord;
231 for (int i=0; i<theta_ord; i++) {
232 theta_nodes[i] = theta_wt * i;
233 }
234
235 // form mesh grid of x-y points, same-r-all-thetas order.
236 sctl::Vector<Real> Xtrg(3 * r_ord * theta_ord);
237 for (int i=0; i<r_ord; i++) {
238 for (int j=0; j<theta_ord; j++) {
239 sctl::Long node_ind = i*theta_ord + j;
240 Xtrg[3*node_ind + 0] = 1.0; // evaluation at x=1.0 only
241 Xtrg[3*node_ind + 1] = 0.5 + r_nodes[i] * sctl::cos<Real>(theta_nodes[j]);
242 Xtrg[3*node_ind + 2] = 0.5 + r_nodes[i] * sctl::sin<Real>(theta_nodes[j]);
243 }
244 }
245
246 // Compute U at meshgrid:
247 sctl::Vector<Real> Utrg;
248 LPO.SetTargetCoord(Xtrg);
249 BIO(&Utrg, sigma);
250 Utrg += bg_poiseuille(Xtrg);
251
252 // Quadrature to integrate
253 Real Q = 0.;
254 for (int i=0; i<r_ord; i++) {
255 for (int j=0; j<theta_ord; j++) {
256 sctl::Long node_ind = i*theta_ord + j;
257 Real u = Utrg[3*node_ind + 0];
258 Q += u * r_nodes[i] * theta_wt * r_wts[i];
259 }
260 }
261
262 Real Q2 = Q * mu / (-dpdx) / Rin / Rin / Rin / Rin; // Q in paper = mu/-dp/R_in^4 * int u_code dA
263
264 return Q2;
265
266 };
